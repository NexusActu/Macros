'================================================================
' MACRO 1 - CopiarNotasMacro1
' Busca col F de Pago a proveedores en col A de Acumulado
' Copia B:F fila 14 de "Copia de notas" y pega en BG:BK
' Pinta rosa si no encuentra
'================================================================
Sub CopiarNotasMacro1()

    ' ============================================================
    ' *** CONFIGURACION - CAMBIA ESTOS VALORES SI ES NECESARIO ***
    Const WB_PROVEEDORES    As String = "Pago a proveedores"  ' Nombre del archivo proveedores
    Const HOJA_PROVEEDORES  As String = "proveedores"         ' Nombre de la hoja en archivo proveedores -- cambia si es necesario
    Const HOJA_NOTAS        As String = "Copia de notas"      ' Nombre de la hoja con los datos a copiar -- cambia si es necesario
    Const FILA_NOTAS        As Long = 14          ' Fila de donde se copia en "Copia de notas"
    Const COL_COPIA_INICIO  As Long = 2           ' Columna inicio de copia en notas (B=2)
    Const COL_COPIA_FIN     As Long = 6           ' Columna fin de copia en notas (F=6)
    Const COL_BUSCAR_ACUM   As Long = 1           ' Columna A de Acumulado donde se busca
    Const COL_BUSCAR_PROV   As Long = 6           ' Columna F de Pago a proveedores
    Const COL_PEGAR_INICIO  As Long = 59          ' Columna BG=59 donde empieza a pegar
    Const FILA_INICIO       As Long = 2           ' Primera fila de datos
    Const COLOR_ROSA        As Long = 16751052    ' RGB(255, 182, 193)
    ' ============================================================

    Dim wbAcum      As Workbook
    Dim wbProv      As Workbook
    Dim wsAcum      As Worksheet
    Dim wsProv      As Worksheet
    Dim wsNotas     As Worksheet
    Dim dictProv    As Object
    Dim filaActual  As Long
    Dim ultimaFila  As Long
    Dim clave       As String

    Set wbAcum  = ThisWorkbook
    Set wsAcum  = wbAcum.ActiveSheet
    Set wsNotas = wbAcum.Sheets(HOJA_NOTAS)

    ' --- Buscar Pago a proveedores entre libros abiertos
    Dim wb As Workbook
    For Each wb In Application.Workbooks
        If InStr(1, wb.Name, WB_PROVEEDORES, vbTextCompare) > 0 Then
            Set wbProv = wb
            Exit For
        End If
    Next wb

    If wbProv Is Nothing Then
        Dim respuesta As Integer
        respuesta = MsgBox("No se encontro el archivo '" & WB_PROVEEDORES & "' abierto." & vbNewLine & _
                           "¿Deseas seleccionarlo manualmente?", _
                           vbYesNo + vbExclamation, "Archivo no encontrado")
        If respuesta = vbNo Then
            MsgBox "Proceso cancelado.", vbInformation
            Exit Sub
        End If

        Dim rutaProv As String
        rutaProv = Application.GetOpenFilename( _
                   "Archivos Excel (*.xlsx; *.xlsm; *.xls), *.xlsx; *.xlsm; *.xls", _
                   , "Selecciona el archivo: " & WB_PROVEEDORES)
        If rutaProv = "False" Then
            MsgBox "No se selecciono ningun archivo. Proceso cancelado.", vbInformation
            Exit Sub
        End If

        Set wbProv = Workbooks.Open(rutaProv)
    End If

    ' --- Apuntar a la hoja correcta de Pago a proveedores
    Set wsProv = wbProv.Sheets(HOJA_PROVEEDORES)

    ' --- Cargar columna F de Pago a proveedores en diccionario
    Set dictProv = CreateObject("Scripting.Dictionary")
    dictProv.CompareMode = vbTextCompare

    Dim ultimaProv As Long
    ultimaProv = wsProv.Cells(wsProv.Rows.Count, COL_BUSCAR_PROV).End(xlUp).Row

    Dim r As Long
    For r = 2 To ultimaProv
        Dim keyProv As String
        keyProv = Trim(CStr(wsProv.Cells(r, COL_BUSCAR_PROV).Value))
        If keyProv <> "" And Not dictProv.Exists(keyProv) Then
            dictProv.Add keyProv, r
        End If
    Next r

    ' --- Recorrer filas de Acumulado
    ultimaFila = wsAcum.Cells(wsAcum.Rows.Count, COL_BUSCAR_ACUM).End(xlUp).Row

    Application.ScreenUpdating = False

    Dim filasCopiadas As Long
    Dim filasSinMatch As Long
    filasCopiadas = 0
    filasSinMatch = 0

    For filaActual = FILA_INICIO To ultimaFila

        clave = Trim(CStr(wsAcum.Cells(filaActual, COL_BUSCAR_ACUM).Value))
        If clave = "" Then GoTo SiguienteFila1

        If dictProv.Exists(clave) Then
            Dim colOrigen  As Long
            Dim colDestino As Long
            colDestino = COL_PEGAR_INICIO

            For colOrigen = COL_COPIA_INICIO To COL_COPIA_FIN
                wsAcum.Cells(filaActual, colDestino).Value = wsNotas.Cells(FILA_NOTAS, colOrigen).Value
                colDestino = colDestino + 1
            Next colOrigen

            filasCopiadas = filasCopiadas + 1
        Else
            wsAcum.Rows(filaActual).Interior.Color = COLOR_ROSA
            filasSinMatch = filasSinMatch + 1
        End If

SiguienteFila1:
    Next filaActual

    Application.ScreenUpdating = True

    MsgBox "Macro 1 completada." & vbNewLine & vbNewLine & _
           "✅ Filas copiadas:          " & filasCopiadas & vbNewLine & _
           "🌸 Filas sin coincidencia: " & filasSinMatch, _
           vbInformation, "Listo"

End Sub
